(dropdown, uitbreidbaar) motortype text not null, monteur text, aantal_koppen int not null default 1, status text not null default 'open', -- open | ter_controle | goedgekeurd | teruggestuurd opmerking_controle text, created_at timestamptz not null default now() ); create table if not exists cilinderkop_rapporten ( id uuid primary key default gen_random_uuid(), werkorder_id uuid not null references werkorders(id) on delete cascade, nummer int not null, disassembly jsonb not null default '[]', seat_ring jsonb not null default '[]', valve jsonb not null default '[]', valve_stem jsonb not null default '[]', valve_guide jsonb not null default '[]', valve_checks jsonb not null default '[]', assembly jsonb not null default '[]', werkuren numeric, aanvullende_info text, updated_at timestamptz not null default now() ); create index if not exists idx_werkorders_werkopdracht on werkorders(werkopdracht_id); create index if not exists idx_cilinderkop_werkorder on cilinderkop_rapporten(werkorder_id); alter table werkopdrachten enable row level security; alter table werkorders enable row level security; alter table cilinderkop_rapporten enable row level security; drop policy if exists "allow all werkorders" on werkorders; create policy "allow all werkopdrachten" on werkopdrachten for all using (true) with check (true); create policy "allow all werkorders" on werkorders for all using (true) with check (true); create policy "allow all cilinderkop_rapporten" on cilinderkop_rapporten for all using (true) with check (true); grant usage on schema public to anon; grant select, insert, update, delete on public.werkopdrachten to anon; grant select, insert, update, delete on public.werkorders to anon; grant select, insert, update, delete on public.cilinderkop_rapporten to anon; NOTIFY pgrst, 'reload schema';
